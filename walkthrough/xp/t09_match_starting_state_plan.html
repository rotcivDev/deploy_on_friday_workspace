T09 Implementation Plan: Create Match Starting State

Intent

Goal: Implement the dtd_match_t game state structure and initialization rules. Define how the game sets up the Encounter and Treasure decks, instantiates the player array, and seeds the game's deterministic RNG.

A multiplayer game needs a unified match state struct. This match state holds the player list, active cards, current phase (e.g. Draw Phase, Play Phase, Combat Phase), current player index, and the match random state, making the complete system state easy to test, serialize, and verify.

#define MAX_PLAYERS 4

typedef struct {
    dtd_player_t players[MAX_PLAYERS];

    dtd_rng_t rng;
    uint64_t seed;

Example Match Implementation

Human-created file: domain/match.c

#include "match.h"
#include <string.h>

bool dtd_match_create(dtd_match_t *match, const dtd_content_definitions_t *content, uint64_t seed, int player_count, const int class_ids[MAX_PLAYERS]) {
    if (!match || !content || player_count < 1 || player_count > MAX_PLAYERS) return false;

    // Validate loaded content definitions
    if (!dtd_validate_content(content)) return false;

    match->player_count = player_count;
    match->current_player_index = 0;
    match->seed = seed;
    match->phase = PHASE_START_TURN;
    match->active_encounter = NULL;

    dtd_rng_init(&match->rng, seed);
    dtd_deck_init(&match->encounter_deck);
    dtd_deck_init(&match->treasure_deck);

    // Load Decks from content definitions
    for (int i = 0; i < content->card_count; i++) {
        const dtd_card_def_t *card = &content->cards[i];
        if (card->type == CARD_MONSTER || card->type == CARD_CURSE || card->type == CARD_AI) {
            for (int k = 0; k < card->quantity; k++) {
                dtd_deck_add_card(&match->encounter_deck, card);
            }
        } else if (card->type == CARD_EQUIPMENT || card->type == CARD_TREASURE) {
            for (int k = 0; k < card->quantity; k++) {
                dtd_deck_add_card(&match->treasure_deck, card);
            }
        }
    }

    // Shuffle Decks
    dtd_deck_shuffle(&match->encounter_deck, &match->rng);
    dtd_deck_shuffle(&match->treasure_deck, &match->rng);

    // Initialize players and draw 2 starting treasure cards to hands
    for (int i = 0; i < player_count; i++) {
        // Find class definition
        int base_power = 0;
        const char *class_name = "Developer";
        for (int c = 0; c < content->class_count; c++) {
            if (content->classes[c].class_id == class_ids[i]) {
                base_power = content->classes[c].base_power_bonus;
                class_name = content->classes[c].name;
                break;
            }
        }

        dtd_player_init(&match->players[i], class_name, base_power);

        // Draw 2 starting treasure cards
        for (int d = 0; d < 2; d++) {
            const dtd_card_def_t *treasure = dtd_deck_draw(&match->treasure_deck, &match->rng);
            if (treasure) {
                dtd_player_draw_to_hand(&match->players[i], treasure);
            }
        }
    }

    return true;
}

Example Match Test

Human-created file: tests/domain/test_match_init.c

#include "match.h"
#include <assert.h>

int main(void) {
    // Setup fake content
    dtd_class_def_t test_classes[6] = {
        {1, "Backend Developer", 1},
        {2, "Frontend Developer", 0},
        {3, "DevOps", 0}, {4, "QA", 0}, {5, "Sec", 0}, {6, "AI", 0}
    };
    dtd_card_def_t test_cards[4] = {
        {"Monster A", CARD_MONSTER, 5, 0, 5},
        {"Equip A", CARD_EQUIPMENT, 0, 2, 5},
        {"Curse A", CARD_CURSE, 0, 0, 5},
        {"AI A", CARD_AI, 2, 0, 5}
    };
    dtd_content_definitions_t content = {
        .classes = test_classes,
        .class_count = 6,
        .cards = test_cards,
        .card_count = 4
    };

    dtd_match_t match;
    int class_selections[3] = {1, 2, 3};
    bool ok = dtd_match_create(&match, &content, 42, 3, class_selections);
    assert(ok);

    assert(match.player_count == 3);
    assert(match.phase == PHASE_START_TURN);
    assert(match.current_player_index == 0);

    // Verify Backend Dev has base power bonus 1, level 1, total power = 2 (no items equipped yet)
    assert(match.players[0].level == 1);
    assert(dtd_player_get_power(&match.players[0]) == 2);

    // Verify starting hands have 2 cards
    assert(match.players[0].hand_count == 2);
    assert(match.players[1].hand_count == 2);

    return 0;
}
