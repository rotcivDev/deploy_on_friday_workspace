T08 Implementation Plan: Build Treasure Deck and Equipment Progression

Intent

Goal: Implement the Treasure deck, card draw mechanics, player hands, and equipping of cards which modify player Power stat calculations.

In Deploy on Friday, players receive rewards (Treasure cards) upon winning combats. These cards can be kept in hand and equipped to permanently increase player combat Power. This plan establishes the card hand tracking and stats modifier application.

Example Player Implementation

Human-created file: domain/player.c

#include "player.h"
#include <string.h>

void game_player_init(game_player_t *player, const char *name, int base_power_bonus) {
    if (!player) return;
    strncpy(player->name, name ? name : "Player", sizeof(player->name) - 1);
    player->level = 1;
    player->base_power_bonus = base_power_bonus;
    player->hand_count = 0;
    player->equipped_count = 0;
}

bool game_player_draw_to_hand(game_player_t *player, const game_card_def_t *card) {
    if (!player || !card || player->hand_count >= MAX_HAND_CARDS) return false;
    player->hand[player->hand_count++] = card;
    return true;
}

bool game_player_equip_from_hand(game_player_t *player, int hand_index) {
    if (!player || hand_index < 0 || hand_index >= player->hand_count) return false;
    if (player->equipped_count >= MAX_EQUIPPED_CARDS) return false;

    const game_card_def_t *card = player->hand[hand_index];
    if (card->type != CARD_EQUIPMENT) return false;

    // Add to equipped
    player->equipped[player->equipped_count++] = card;

    // Remove from hand and shift remaining hand cards left
    for (int i = hand_index; i < player->hand_count - 1; i++) {
        player->hand[i] = player->hand[i + 1];
    }
    player->hand_count--;
    return true;
}

int game_player_get_power(const game_player_t *player) {
    if (!player) return 0;
    int power = player->level + player->base_power_bonus;
    for (int i = 0; i < player->equipped_count; i++) {
        power += player->equipped[i]->power_bonus;
    }
    return power;
}

Example Equipment Test

Human-created file: tests/domain/test_equipment.c

#include "player.h"
#include "content.h"
#include <assert.h>

int main(void) {
    game_player_t player;
    game_player_init(&player, "Senior backend devs", 1); // level 1 + base 1 = 2

    assert(game_player_get_power(&player) == 2);

    // Create ThinkPad
    game_card_def_t thinkpad = {"ThinkPad T480", CARD_EQUIPMENT, 0, 2, 1};

    // Draw and equip
    assert(game_player_draw_to_hand(&player, &thinkpad));
    assert(player.hand_count == 1);
    
    assert(game_player_equip_from_hand(&player, 0));
    assert(player.hand_count == 0);
    assert(player.equipped_count == 1);

    // Power should now be level (1) + base (1) + ThinkPad (2) = 4
    assert(game_player_get_power(&player) == 4);

    return 0;
}
