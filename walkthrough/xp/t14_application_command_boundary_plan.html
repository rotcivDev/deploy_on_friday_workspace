T14 Implementation Plan: Add Application Command Boundary

Intent

Goal: Implement the game_game_service_t API wrapper. Route all match changes through explicit application commands, validating phase conditions before modifying the underlying game state.

In Clean Architecture, client applications and bot scripts must not call domain structures directly to change levels or draw cards. All moves must pass through an Application Service command route. The service evaluates if the action is legal for the current phase, execute the move, and return a result.

Implementation Story

Source backlog item: T14, A4, AR5, SP3.

Domain behavior served: Enforcing state phase transitions, verifying card ownership in hand before playing/equipping, and resolving turn cycles.

User-visible outcome served: Ensure players only make legal moves, preventing cheating or accidental double-actions during a turn.

Boundaries touched by the human: application/game_service.h, application/game_service.c, tests/application/test_game_service.c.

Out of scope: Live socket networking or user account authorization. Keep the service local.

Application Command Flow

The service must support the following command dispatches:

  • CMD_START_MATCH: Transitions state to PHASE_START_TURN.
  • CMD_KICK_BACKLOG (Draw Encounter): Allowed during PHASE_START_TURN. Draws top Encounter card. Sets active encounter. Transitions to PHASE_RESOLVE_ENCOUNTER.
  • CMD_RESOLVE_ENCOUNTER: Triggers combat or curse resolution. Transitions to PHASE_LOBBY_PLAY.
  • CMD_EQUIP_CARD: Allowed during PHASE_LOBBY_PLAY. Equips card from active player's hand.
  • CMD_END_TURN: Allowed during PHASE_LOBBY_PLAY. Advances current player index. Transitions back to PHASE_START_TURN.

#ifndef GAME_GAME_SERVICE_H
#define GAME_GAME_SERVICE_H

#include "match.h"
#include "command_result.h"

typedef enum {
    CMD_START_MATCH,
    CMD_KICK_BACKLOG,
    CMD_RESOLVE_ENCOUNTER,
    CMD_EQUIP_CARD,
    CMD_END_TURN
} game_command_type_t;

typedef struct {
    game_command_type_t type;
    int player_index;
    int hand_index;
} game_command_t;

typedef struct {
    game_match_t match;
    bool match_active;
} game_game_service_t;

void game_service_init(game_game_service_t *svc);
game_command_result_t game_service_execute(game_game_service_t *svc, const game_command_t *cmd, const game_content_definitions_t *content);

#endif

Example Game Service Implementation

Human-created file: application/game_service.c

#include "game_service.h"
#include "combat.h"
#include "curse.h"
#include <stdio.h>

void game_service_init(game_game_service_t *svc) {
    if (svc) {
        svc->match_active = false;
    }
}

game_command_result_t game_service_execute(game_game_service_t *svc, const game_command_t *cmd, const game_content_definitions_t *content) {
    game_command_result_t result;
    game_result_init(&result, CMD_OK);

    if (!svc || !cmd) {
        result.status = CMD_ERROR;
        return result;
    }

    if (cmd->type == CMD_START_MATCH) {
        int classes[3] = {1, 2, 3}; // Stub starting classes
        if (game_match_create(&svc->match, content, 12345, 3, classes)) {
            svc->match_active = true;
            game_result_add_event(&result, EV_MATCH_STARTED, "Match started successfully", 0);
            result.status = CMD_OK;
        } else {
            result.status = CMD_ERROR;
        }
        return result;
    }

    if (!svc->match_active) {
        result.status = CMD_INVALID_ACTION;
        return result;
    }

    game_match_t *match = &svc->match;

    switch (cmd->type) {
        case CMD_KICK_BACKLOG:
            if (match->phase != PHASE_START_TURN) {
                result.status = CMD_INVALID_PHASE;
                break;
            }
            // Draw encounter card
            match->active_encounter = game_deck_draw(&match->encounter_deck, &match->rng);
            if (match->active_encounter) {
                char buffer[128];
                snprintf(buffer, sizeof(buffer), "Drew encounter card: %s", match->active_encounter->name);
                game_result_add_event(&result, EV_CARD_DRAWN, buffer, match->current_player_index);
                match->phase = PHASE_RESOLVE_ENCOUNTER;
                result.status = CMD_OK;
            } else {
                result.status = CMD_ERROR;
            }
            break;

        case CMD_RESOLVE_ENCOUNTER:
            if (match->phase != PHASE_RESOLVE_ENCOUNTER || !match->active_encounter) {
                result.status = CMD_INVALID_PHASE;
                break;
            }
            
            if (match->active_encounter->type == CARD_MONSTER) {
                game_combat_outcome_t outcome = game_resolve_combat(match, match->current_player_index, match->active_encounter);
                if (outcome.win) {
                    game_result_add_event(&result, EV_COMBAT_WON, "Player won combat", match->current_player_index);
                } else {
                    game_result_add_event(&result, EV_COMBAT_LOST, "Player lost combat (bad stuff applied)", match->current_player_index);
                }
            } else if (match->active_encounter->type == CARD_CURSE) {
                game_apply_curse(match, match->current_player_index, match->active_encounter);
                game_result_add_event(&result, EV_CURSE_APPLIED, "Curse effect applied", match->current_player_index);
            }

            // Put active encounter to discard pile
            game_deck_discard(&match->encounter_deck, match->active_encounter);
            match->active_encounter = NULL;
            match->phase = PHASE_LOBBY_PLAY;
            result.status = CMD_OK;
            break;

        case CMD_EQUIP_CARD:
            if (match->phase != PHASE_LOBBY_PLAY) {
                result.status = CMD_INVALID_PHASE;
                break;
            }
            if (game_player_equip_from_hand(&match->players[match->current_player_index], cmd->hand_index)) {
                game_result_add_event(&result, EV_CARD_EQUIPPED, "Equipped item card", match->current_player_index);
                result.status = CMD_OK;
            } else {
                result.status = CMD_INVALID_ACTION;
            }
            break;

        case CMD_END_TURN:
            if (match->phase != PHASE_LOBBY_PLAY) {
                result.status = CMD_INVALID_PHASE;
                break;
            }
            match->current_player_index = (match->current_player_index + 1) % match->player_count;
            match->phase = PHASE_START_TURN;
            game_result_add_event(&result, EV_TURN_ENDED, "Turn advanced to next developer", match->current_player_index);
            result.status = CMD_OK;
            break;

        default:
            result.status = CMD_INVALID_ACTION;
            break;
    }

    return result;
}

Example Game Service Test

Human-created file: tests/application/test_game_service.c

#include "game_service.h"
#include <assert.h>

int main(void) {
    game_class_def_t test_classes[6] = {
        {1, "QA", 0}, {2, "QA", 0}, {3, "QA", 0}, {4, "QA", 0}, {5, "QA", 0}, {6, "QA", 0}
    };
    game_card_def_t test_cards[4] = {
        {"Monster A", CARD_MONSTER, 3, 0, 5},
        {"Equip A", CARD_EQUIPMENT, 0, 2, 5},
        {"Curse A", CARD_CURSE, 0, 0, 5},
        {"AI A", CARD_AI, 2, 0, 5}
    };
    game_content_definitions_t content = {
        .classes = test_classes,
        .class_count = 6,
        .cards = test_cards,
        .card_count = 4
    };

    game_game_service_t svc;
    game_service_init(&svc);

    // Verify resolve before start fails
    game_command_t cmd_resolve = {CMD_RESOLVE_ENCOUNTER, 0, 0};
    game_command_result_t res = game_service_execute(&svc, &cmd_resolve, &content);
    assert(res.status == CMD_INVALID_ACTION);

    // Start match
    game_command_t cmd_start = {CMD_START_MATCH, 0, 0};
    res = game_service_execute(&svc, &cmd_start, &content);
    assert(res.status == CMD_OK);

    // Draw encounter
    game_command_t cmd_draw = {CMD_KICK_BACKLOG, 0, 0};
    res = game_service_execute(&svc, &cmd_draw, &content);
    assert(res.status == CMD_OK);

    // Verify resolve now works
    res = game_service_execute(&svc, &cmd_resolve, &content);
    assert(res.status == CMD_OK);

    return 0;
}
